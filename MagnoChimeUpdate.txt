#!/bin/bash
export DEBIAN_FRONTEND=noninteractive

# 1. Disable harsh hardware & terminal beeps permanently
sudo -n rmmod pcspkr 2>/dev/null || true
echo "blacklist pcspkr" | sudo -n tee /etc/modprobe.d/blacklist-pcspkr.conf > /dev/null
xset b off 2>/dev/null || true

grep -q "bell-style none" ~/.inputrc 2>/dev/null || echo "set bell-style none" >> ~/.inputrc
sudo -n mkdir -p /etc/skel
grep -q "bell-style none" /etc/skel/.inputrc 2>/dev/null || echo "set bell-style none" | sudo -n tee -a /etc/skel/.inputrc > /dev/null

# 2. Synthesize custom warm lo-fi MagnoOS chimes into /usr/share/sounds/MagnoOS
sudo -n mkdir -p /usr/share/sounds/MagnoOS/stereo

sudo -n python3 -c '
import os, wave, math, struct

OUT_DIR = "/usr/share/sounds/MagnoOS/stereo"
RATE = 44100

def write_chime(filename, notes, duration=0.85, decay=4.5, vol=0.28):
    path = os.path.join(OUT_DIR, filename)
    n_samples = int(RATE * duration)
    buf = bytearray()
    for i in range(n_samples):
        t = i / float(RATE)
        # Soft attack (15ms) + warm exponential decay
        attack = min(1.0, t / 0.015)
        env = attack * math.exp(-decay * t)
        val = 0.0
        for idx, (freq, delay) in enumerate(notes):
            if t >= delay:
                lt = t - delay
                l_att = min(1.0, lt / 0.015)
                l_env = l_att * math.exp(-decay * lt)
                # Fundamental + subtle warm 2nd harmonic
                val += l_env * (math.sin(2 * math.pi * freq * lt) + 0.18 * math.sin(4 * math.pi * freq * lt))
        sample = max(-32767, min(32767, int(val * vol * 32767)))
        buf += struct.pack("<hh", sample, sample)
    with wave.open(path, "w") as wf:
        wf.setnchannels(2)
        wf.setsampwidth(2)
        wf.setframerate(RATE)
        wf.writeframes(bytes(buf))

# Warm Fmaj9 ambient chord for Startup (F4, A4, C5, E5, G5)
write_chime("desktop-login.wav", [(349.23, 0.0), (440.00, 0.09), (523.25, 0.18), (659.25, 0.27), (783.99, 0.36)], duration=1.6, decay=2.6, vol=0.25)

# Soft two-note bell for Notifications (C5 -> G5)
write_chime("message-new-instant.wav", [(523.25, 0.0), (783.99, 0.11)], duration=0.75, decay=5.2, vol=0.24)
write_chime("bell.wav", [(587.33, 0.0), (880.00, 0.08)], duration=0.55, decay=6.5, vol=0.20)

# Gentle rising chime for Charger / USB Plugged In (G4 -> B4 -> D5)
write_chime("power-plug.wav", [(392.00, 0.0), (493.88, 0.07), (587.33, 0.14)], duration=0.65, decay=5.5, vol=0.22)

# Gentle falling chime for Unplugged (D5 -> G4)
write_chime("power-unplug.wav", [(587.33, 0.0), (392.00, 0.09)], duration=0.60, decay=5.5, vol=0.20)

# Crisp soft droplet pop for Screenshots (A5 -> E6)
write_chime("screen-capture.wav", [(880.00, 0.0), (1318.51, 0.04)], duration=0.35, decay=9.0, vol=0.22)

# Mellow low double-pulse instead of harsh error buzz (A3 -> F3)
write_chime("dialog-warning.wav", [(220.00, 0.0), (174.61, 0.14)], duration=0.60, decay=5.5, vol=0.24)
'

# 3. Create the MagnoOS Sound Theme index
sudo -n tee /usr/share/sounds/MagnoOS/index.theme > /dev/null << 'EOF'
[Sound Theme]
Name=MagnoOS Chill
Comment=Warm ambient chimes for MagnoOS
Directories=stereo

[stereo]
OutputProfile=stereo
EOF

sudo -n chmod -R 644 /usr/share/sounds/MagnoOS/stereo/*.wav /usr/share/sounds/MagnoOS/index.theme
sudo -n chmod 755 /usr/share/sounds/MagnoOS /usr/share/sounds/MagnoOS/stereo

# 4. Activate MagnoOS Chill Sound Theme in Cinnamon & GSettings
gsettings set org.cinnamon.desktop.sound theme-name 'MagnoOS' 2>/dev/null || true
gsettings set org.cinnamon.desktop.sound event-sounds true 2>/dev/null || true
gsettings set org.cinnamon.desktop.sound input-feedback-sounds false 2>/dev/null || true

# 5. Wire startup chime into autostart & /etc/skel
mkdir -p ~/.config/autostart
sudo -n mkdir -p /etc/skel/.config/autostart

cat << 'EOF' > ~/.config/autostart/magno-chime.desktop
[Desktop Entry]
Type=Application
Name=MagnoOS Startup Chime
Exec=aplay -q /usr/share/sounds/MagnoOS/stereo/desktop-login.wav
X-GNOME-Autostart-enabled=true
EOF
sudo -n cp ~/.config/autostart/magno-chime.desktop /etc/skel/.config/autostart/magno-chime.desktop

# 6. Play the new startup chord right away so you can hear it!
aplay -q /usr/share/sounds/MagnoOS/stereo/desktop-login.wav 2>/dev/null || true
